-- ===================================================
-- 1. CAFE BASIC METRICS (REVENUE, COST & NET PROFIT)
-- ===================================================
SELECT M.ITEM_NAME,M.CATEGORY,
SUM(S.QUANTITY_SOLD) AS TOTAL_SPENT,
SUM (S.QUANTITY_SOLD * M.SELLING_PRICE) AS TOTAL_REVENUE,
SUM (S.QUANTITY_SOLD *(M.SELLING_PRICE - M.MAKING_COST)) AS NET_PROFIT
FROM CAFE_MENU M 
JOIN CAFE_SALES S ON M.ITEM_ID=S.ITEM_ID
GROUP BY M.ITEM_NAME,M.CATEGORY;


-- ===================================================
-- 2. CAFE PROFITABILITY CATEGORIZATION (CASE WHEN)
-- ===================================================
SELECT M.ITEM_NAME,M.CATEGORY,
SUM (S.QUANTITY_SOLD * (M.SELLING_PRICE - M.MAKING_COST)) AS NET_PROFIT,
CASE
WHEN SUM(S.QUANTITY_SOLD * (M.SELLING_PRICE - M.MAKING_COST)) > 159 THEN 'Super Profit'
WHEN SUM(S.QUANTITY_SOLD * (M.SELLING_PRICE - M.MAKING_COST))BETWEEN 99 AND 159 THEN 'AVG PROFIT'
ELSE 'LOW MARGIN'
END AS PROFIT_CONTRI
FROM CAFE_MENU M 
JOIN CAFE_SALES S ON M.ITEM_ID=S.ITEM_ID
GROUP BY M.ITEM_NAME,M.CATEGORY;
